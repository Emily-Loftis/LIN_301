Cleaning the text and building the table

rows = []
speaker = ""
line_number = 0

for line in lines:
    line = line.strip()

    if line != "":
        if ": " in line:
            parts = line.split(": ")
            speaker = parts[0]
            line = ": ".join(parts[1:])

        if speaker != "MUSIC":
            line_number += 1
            rows.append({
                "line": line,
                "speaker": speaker,
                "line_number": line_number
            })

df = pd.DataFrame(rows)

print(len(df))
