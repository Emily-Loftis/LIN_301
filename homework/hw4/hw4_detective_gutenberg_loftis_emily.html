Part 1 — Setup and Cleaning

import re
from collections import Counter
import matplotlib.pyplot as plt

try:
    with open("pride_and_prejudice.txt", encoding="utf-8") as f:
        book1_lines = f.readlines()

    with open("frankenstein.txt", encoding="utf-8") as f:
        book2_lines = f.readlines()

except FileNotFoundError as error:
    print("Could not find a book file:", error)
    book1_lines = []
    book2_lines = []

for lines in [book1_lines, book2_lines]:
    while lines and not lines[0].startswith("*** START OF"):
        lines.pop(0)
    if lines:
        lines.pop(0)

    while lines and not lines[-1].startswith("*** END OF"):
        lines.pop()
    if lines:
        lines.pop()

book1_text = "".join(book1_lines).lower()
book2_text = "".join(book2_lines).lower()

book1_words = [w for w in re.split(r"[\W]+", book1_text) if w]
book2_words = [w for w in re.split(r"[\W]+", book2_text) if w]

print("Pride and Prejudice:", len(book1_words), "words")
print("Frankenstein:", len(book2_words), "words")
Pride and Prejudice: 128755 words
Frankenstein: 75379 words

My code reads both books and removes the Gutenberg information at the beginning and end using while loops. Then it puts the lines together, makes everything lowercase, and splits the text into words. It also removes any empty strings. I used ChatGPT to troubleshoot a few things.
